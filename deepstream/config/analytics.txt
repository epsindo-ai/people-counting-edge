# People counting lines (nvdsanalytics).
# Coordinates are in config-width x config-height pixels (same as streammux).
# Line names (line-crossing-<name>) must match entry-lines / exit-lines in msgconv.txt.

[property]
enable=1
config-width=1920
config-height=1080
# 2 = draw lines, counts and object status on video
osd-mode=2
display-font-size=12

# line-crossing-<name>=x1;y1;x2;y2;x3;y3;x4;y4
#   x1;y1 -> x2;y2  direction arrow: a crossing counts only when moving this way
#   x3;y3 -> x4;y4  the line people cross
# Entry and Exit share one line with opposite arrows.

# Camera 0 (first enabled source). Line: x=950, y 150..450.
[line-crossing-stream-0]
enable=1
# 0 = person
class-id=0
# Entry: moving left
line-crossing-Entry=1050;325;850;325;950;150;950;450
# Exit: moving right
line-crossing-Exit=850;325;1050;325;950;150;950;450
mode=strict
extended=0
